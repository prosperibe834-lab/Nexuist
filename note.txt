document.addEventListener("DOMContentLoaded", () => {

    // All animation selectors
    const animatedElements = document.querySelectorAll(`
        .st-animate,
        .at-animate,
        .nx-animate,
        .sp-animate,
        .dm-animate,
        .cs-animate,
        .ts-animate,
        .ln-animate
    `);

    // Reusable observer
    const globalObserver = new IntersectionObserver((entries, observer) => {

        entries.forEach(entry => {

            if (entry.isIntersecting) {

                // Activate animation
                entry.target.classList.add("active");

                ////////////////////////////////////////////////////////////
                // NX COUNTERS
                ////////////////////////////////////////////////////////////
                if (
                    entry.target.classList.contains('nx-stats-bar') &&
                    !entry.target.classList.contains('counted')
                ) {

                    entry.target.classList.add('counted');

                    const counters = document.querySelectorAll('.nx-counter');

                    counters.forEach(counter => {

                        const target = +counter.getAttribute('data-target');
                        const increment = target / 50;

                        const updateCount = () => {

                            const current = +counter.innerText;

                            if (current < target) {

                                counter.innerText = Math.ceil(current + increment);

                                setTimeout(updateCount, 30);

                            } else {

                                counter.innerText = target;
                            }
                        };

                        updateCount();
                    });
                }

                // Stop observing after animation
                observer.unobserve(entry.target);
            }

        });

    }, {
        threshold: 0.12
    });

    // Observe all elements
    animatedElements.forEach((el, index) => {

        // Auto stagger delay
        el.style.transitionDelay = `${index * 0.05}s`;

        globalObserver.observe(el);
    });

});

window.addEventListener("resize", () => {

    const canvas = document.getElementById('at-live-chart');

    if (!canvas) return;

    const ctx = canvas.getContext('2d');

    canvas.width = canvas.parentElement.offsetWidth;
    canvas.height = 350;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    ctx.beginPath();
    ctx.strokeStyle = '#0070f3';
    ctx.lineWidth = 3;
    ctx.moveTo(0, 200);

    for (let i = 0; i < canvas.width; i += 20) {

        const y = 150 + Math.sin(i * 0.05) * 40 + Math.random() * 30;

        ctx.lineTo(i, y);
    }

    ctx.stroke();

    ctx.lineTo(canvas.width, canvas.height);
    ctx.lineTo(0, canvas.height);

    const grad = ctx.createLinearGradient(0, 0, 0, 400);

    grad.addColorStop(0, 'rgba(0, 112, 243, 0.2)');
    grad.addColorStop(1, 'transparent');

    ctx.fillStyle = grad;
    ctx.fill();
});

document.addEventListener("click", (e) => {

    const navMenu = document.getElementById("nav-menu");
    const menuToggle = document.getElementById("mobile-menu");
    const navbar = document.getElementById("navbar");

    if (!navMenu || !menuToggle || !navbar) return;

    const clickedInsideMenu = navMenu.contains(e.target);
    const clickedToggle = menuToggle.contains(e.target);

    if (
        navMenu.classList.contains("is-open") &&
        !clickedInsideMenu &&
        !clickedToggle
    ) {

        navMenu.classList.remove("is-open");
        navbar.classList.remove("menu-is-active");
        menuToggle.classList.remove("is-active");

        document.body.style.overflow = "initial";
    }
});

// SMOOTH SCROLL FOR INTERNAL LINKS
////////////////////////////////////////////////////////////

document.querySelectorAll('a[href^="#"]').forEach(anchor => {

    anchor.addEventListener('click', function (e) {

        const target = document.querySelector(this.getAttribute('href'));

        if (target) {

            e.preventDefault();

            target.scrollIntoView({
                behavior: 'smooth',
                block: 'start'
            });
        }
    });
});

document.querySelectorAll(`
    .io-card,
    .nx-card,
    .sp-step-card,
    .cs-card,
    .ts-card,
    .ln-card,
    .dm-benefit-tile
`).forEach(card => {

    card.addEventListener('mousemove', (e) => {

        const rect = card.getBoundingClientRect();

        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;

        card.style.setProperty('--mouse-x', `${x}px`);
        card.style.setProperty('--mouse-y', `${y}px`);
    });

});
